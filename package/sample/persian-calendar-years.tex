% Jalali calendars for a range of years, nine years to an A4 page, in the manner of calxxxx-yyyy:
% each year is four seasons of three months, each month a column per week and a row per weekday.
% Set the years below. Build: make test   (XeLaTeX; xepersian and the font Parisa are needed)
\documentclass{article}

\usepackage[a4paper,margin=5mm]{geometry}
\usepackage{tikz}
\usetikzlibrary{calendar}
\usepackage{persian-calendar}
\usepackage{xepersian}

\settextfont{Parisa}
\pagestyle{empty}

\newcommand\FirstYear{1405}
\newcommand\LastYear{1413}

% One year: the title (year and its animal), then four rows of seasons, three months each, right to left.
% A month is six week columns plus the column of initials: 7 x 0.3cm = 2.1cm; three years fit the width.
\newcommand\JalaliYear[1]{%
  \begin{tikzpicture}[font=\tiny]
    \node[anchor=base east, font=\footnotesize\bfseries] at (0.3,0.35) {#1\quad\pcalyearname{#1}};
    \foreach \m in {1,...,12} {
      \pgfmathtruncatemacro\col{mod(\m-1,3)}%
      \pgfmathtruncatemacro\row{div(\m-1,3)}%
      \calendar at (-\col*2.15cm, -0.55cm-\row*2.1cm)
        [jalali dates=#1-\m-01 to #1-\m-last, jalali week columns, jalali weekday initials,
         day xshift=-0.3cm, day yshift=0.235cm, jalali month label above centered,
         month text={\pcalmonthname{\pcalcurrentmonth}},
         every jalali weekday/.style={gray}]
        if (Friday) [red];
    }
  \end{tikzpicture}}

\begin{document}
\noindent
\foreach \yr [count=\i] in {\FirstYear,...,\LastYear} {%
  \JalaliYear{\yr}%
  \pgfmathtruncatemacro\n{mod(\i,3)}%
  \ifnum\n=0 \par\bigskip\noindent\else\hfill\fi
}
\end{document}
